import java.util.Scanner;

public class EasyPath {
    static int R, C;
    static int[][] memo;
    static int[][] dirs = {{1,0},{-1,0},{0,1},{0,-1}}; // down, up, right, left

    public static int longestIncreasingPath(int[][] matrix) {
        R = matrix.length;
        C = matrix[0].length;
        memo = new int[R][C];
        int longest = 0;

        for(int i=0; i<R; i++){
            for(int j=0; j<C; j++){
                longest = Math.max(longest, dfs(matrix, i, j));
            }
        }
        return longest;
    }

    static int dfs(int[][] matrix, int i, int j){
        if(memo[i][j] != 0) return memo[i][j];
        int best = 1; // at least this cell
        for(int[] d : dirs){
            int ni = i + d[0], nj = j + d[1];
            if(ni>=0 && nj>=0 && ni<R && nj<C && matrix[ni][nj] > matrix[i][j]){
                best = Math.max(best, 1 + dfs(matrix, ni, nj));
            }
        }
        memo[i][j] = best;
        return best;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int R = sc.nextInt();
        int C = sc.nextInt();
        int[][] matrix = new int[R][C];

        for(int i=0; i<R; i++){
            for(int j=0; j<C; j++){
                matrix[i][j] = sc.nextInt();
            }
        }
        sc.close();

        System.out.println(longestIncreasingPath(matrix));
    }
}
