#include <iostream>
using namespace std;

int main() {
    int row, col;
    cin >> row >> col;

    int a[row][col];

    for(int i = 0; i < row; i++) {
        for(int j = 0; j < col; j++) {
            cin >> a[i][j];
        }
    }

    for(int j = 0; j < col; j++) {
        int max = a[0][j];

        for(int i = 1; i < row; i++) {
            if(a[i][j] > max)
                max = a[i][j];
        }

        cout << "Column " << j + 1 << " Max = " << max << endl;
    }

    return 0;
}
