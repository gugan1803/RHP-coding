a-z and A-Z
#include <stdio.h>
#include <stdlib.h>

int main() {
    char str[1000001];
    scanf("%s", str);

    int flag = 0;

    for (int i = 0; str[i]; i++) {
        char ch = tolower(str[i]);

        if (ch >= 'a' && ch <= 'z') {
            flag |= (1 << (ch - 'a'));
        }
    }

    printf((flag == (1 << 26) - 1) ? "Yes" : "No");

    return 0;
}
